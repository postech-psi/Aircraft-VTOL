cmake_minimum_required(VERSION 3.16)
project(psi_aerodynamics LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(psi-aerodynamics SHARED AeroSystem.cc)
target_compile_features(psi-aerodynamics PRIVATE cxx_std_17)
target_compile_options(psi-aerodynamics PRIVATE -Wall -Wextra)
target_link_libraries(psi-aerodynamics PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)
